```js
function md(r){if(!r)return 0;return 1+Math.max(md(r.left),md(r.right));}
```

### E4. Invert Tree

<details><summary>Solution (JS)</summary>

```js
function inv(r){if(!r)return null;[r.left,r.right]=[inv(r.right),inv(r.left)];return r;}
```

### E6. Check Balanced

<details><summary>Solution (JS)</summary>

```js
function bal(r){function h(n){if(!n)return 0;const l=h(n.left),r2=h(n.right);if(l===-1||r2===-1||Math.abs(l-r2)>1)return -1;return 1+Math.max(l,r2);}return h(r)!==-1;}
```

</details>

<details><summary>Solution (JS)</summary>

