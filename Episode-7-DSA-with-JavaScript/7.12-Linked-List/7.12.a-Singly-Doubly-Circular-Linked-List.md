## Node definition

```
Singly linked list:
  ┌──────┬──────┐    ┌──────┬──────┐    ┌──────┬──────┐
  │ val  │ next │───▶│ val  │ next │───▶│ val  │ next │───▶ null
  └──────┴──────┘    └──────┴──────┘    └──────┴──────┘
    head                                   tail
```

### JavaScript

```js
class ListNode {
    constructor(val, next = null) {
        this.val = val;
        this.next = next;
    }
}
```

### C++

```cpp
struct ListNode {
    int val;
    ListNode* next;
    ListNode(int v, ListNode* n = nullptr) : val(v), next(n) {}
};
```

## Build & traverse

```js
function buildList(arr) {
    let head = null;
    for (let i = arr.length - 1; i >= 0; i--)
        head = new ListNode(arr[i], head);
    return head;
}

function printList(head) {
    let curr = head;
    const parts = [];
    while (curr) { parts.push(curr.val); curr = curr.next; }
    console.log(parts.join(" → ") + " → null");
}
```

```cpp
ListNode* buildList(vector<int>& arr) {
    ListNode* head = nullptr;
    for (int i = arr.size()-1; i >= 0; i--)
        head = new ListNode(arr[i], head);
    return head;
}
```

## Reverse linked list

```
Before: 1 → 2 → 3 → null
After:  3 → 2 → 1 → null

Step-by-step:
  prev=null, curr=1
  next=2, curr.next=null, prev=1, curr=2
  next=3, curr.next=1,   prev=2, curr=3
  next=null, curr.next=2, prev=3, curr=null
  return prev → 3 → 2 → 1 → null
```

```js
function reverse(head) {
    let prev = null, curr = head;
    while (curr) {
        const next = curr.next;
        curr.next = prev;
        prev = curr;
        curr = next;
    }
    return prev;
}
```

```cpp
ListNode* reverse(ListNode* head) {
    ListNode* prev = nullptr;
    while (head) {
        ListNode* next = head->next;
        head->next = prev;
        prev = head;
        head = next;
    }
    return prev;
}
```

## Detect cycle (Floyd's algorithm)

```
  1 → 2 → 3 → 4 → 5
               ↑       ↓
               8 ← 7 ← 6

Fast pointer moves 2 steps, slow moves 1.
If cycle exists, they will meet.
```

```js
function hasCycle(head) {
    let slow = head, fast = head;
    while (fast && fast.next) {
        slow = slow.next;
        fast = fast.next.next;
        if (slow === fast) return true;
    }
    return false;
}
```

### Find cycle start

```js
function findCycleStart(head) {
    let slow = head, fast = head;
    while (fast && fast.next) {
        slow = slow.next;
        fast = fast.next.next;
        if (slow === fast) {
            slow = head;
            while (slow !== fast) { slow = slow.next; fast = fast.next; }
            return slow;
        }
    }
    return null;
}
```

## Merge two sorted lists

```js
function mergeTwoLists(l1, l2) {
    const dummy = new ListNode(0);
    let curr = dummy;
    while (l1 && l2) {
        if (l1.val <= l2.val) { curr.next = l1; l1 = l1.next; }
        else { curr.next = l2; l2 = l2.next; }
        curr = curr.next;
    }
    curr.next = l1 || l2;
    return dummy.next;
}
```

## Remove nth from end

```js
function removeNthFromEnd(head, n) {
    const dummy = new ListNode(0, head);
    let fast = dummy, slow = dummy;
    for (let i = 0; i <= n; i++) fast = fast.next;
    while (fast) { fast = fast.next; slow = slow.next; }
    slow.next = slow.next.next;
    return dummy.next;
}
```

## Doubly linked list

```
  null ← ┌──────┬──────┬──────┐ ⇄ ┌──────┬──────┬──────┐ ⇄ ┌──────┬──────┬──────┐ → null
         │ prev │ val  │ next │   │ prev │ val  │ next │   │ prev │ val  │ next │
         └──────┴──────┴──────┘   └──────┴──────┴──────┘   └──────┴──────┴──────┘
```

```js
class DNode {
    constructor(val, prev = null, next = null) {
        this.val = val;
        this.prev = prev;
        this.next = next;
    }
}
```

## Circular linked list

```
  1 → 2 → 3 → 4
  ↑             │
  └─────────────┘
```

Last node's `next` points back to head.

---

## Operations complexity

| Operation | Singly | Doubly |
|-----------|--------|--------|
| Access by index | O(n) | O(n) |
| Insert at head | O(1) | O(1) |
| Insert at tail | O(n)* | O(1)** |
| Delete at head | O(1) | O(1) |
| Delete at tail | O(n) | O(1) |
| Search | O(n) | O(n) |

*O(1) if tail pointer maintained **with tail pointer
